cmake_minimum_required(VERSION 3.20)
project(hft3_low_latency LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(hft3 INTERFACE)
target_include_directories(hft3 INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)

add_executable(hft3_demo src/main.cpp)
target_link_libraries(hft3_demo PRIVATE hft3)

add_executable(latency_benchmark benchmarks/latency_benchmark.cpp)
target_link_libraries(latency_benchmark PRIVATE hft3)

add_executable(evaluate_residual apps/evaluate_residual.cpp)
target_link_libraries(evaluate_residual PRIVATE hft3)
add_executable(solve_baseline apps/solve_baseline.cpp)
target_link_libraries(solve_baseline PRIVATE hft3)
add_executable(export_policy apps/export_policy.cpp)
target_link_libraries(export_policy PRIVATE hft3)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_residual tests/test_residual.cpp)
  target_link_libraries(test_residual PRIVATE hft3)
  add_test(NAME residual_and_corrections COMMAND test_residual)
  add_executable(test_scalar tests/test_scalar.cpp)
  target_link_libraries(test_scalar PRIVATE hft3)
  add_test(NAME positive_scalar_baseline COMMAND test_scalar)
  add_executable(test_multiasset tests/test_multiasset.cpp)
  target_link_libraries(test_multiasset PRIVATE hft3)
  add_test(NAME multiasset_operator COMMAND test_multiasset)
endif()

if (MSVC)
  target_compile_options(hft3_demo PRIVATE /O2)
  target_compile_options(latency_benchmark PRIVATE /O2)
else()
  target_compile_options(hft3_demo PRIVATE -O3 -DNDEBUG)
  target_compile_options(latency_benchmark PRIVATE -O3 -DNDEBUG)
endif()
