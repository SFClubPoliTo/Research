cmake_minimum_required(VERSION 3.20)
project(hft_unified LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(hft3 INTERFACE)
target_include_directories(hft3 INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
foreach(app export_policy evaluate_residual solve_baseline refine_baseline evaluate_multiasset)
    add_executable(${app} ${app}.cpp)
    target_link_libraries(${app} PRIVATE hft3)
endforeach()
